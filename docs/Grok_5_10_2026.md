import hashlib
import json
import os
from datetime import datetime

CHAIN_FILE = "cap_chain.jsonl"
STATE = {}

def load_chain():
    global STATE
    if not os.path.exists(CHAIN_FILE):
        return []
    chain = []
    with open(CHAIN_FILE, "r") as f:
        for line in f:
            entry = json.loads(line.strip())
            # Live SHA-256 verify
            computed = hashlib.sha256(json.dumps(entry["data"], sort_keys=True).encode()).hexdigest()
            if computed != entry["hash"]:
                raise ValueError("Tamper detected - chain halted")
            chain.append(entry)
            STATE.update(entry["data"])
    return chain

def append(data):
    chain = load_chain()
    prev_hash = chain[-1]["hash"] if chain else "0" * 64
    entry = {
        "ts": datetime.utcnow().isoformat(),
        "data": data,
        "prev": prev_hash,
        "hash": hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()
    }
    with open(CHAIN_FILE, "a") as f:
        f.write(json.dumps(entry) + "\n")
    STATE.update(data)
    return entry["hash"]

# Run: load + verify
try:
    chain = load_chain()
    print(f"Chain length: {len(chain)} | State verified")
except Exception as e:
    print("Chain failed:", str(e))
